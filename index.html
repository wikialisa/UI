import React from 'react';
import { View, Text, TouchableOpacity, StyleSheet } from 'react-native';
import { colors, spacing, radius, typography } from './tokens';

interface PumpKpiCardProps {
  pumpName: string;
  solarKw: number;
  flowRate: number;
  status: 'ACTIVE_PUMPING' | 'OFFLINE';
  onPressAction?: () => void;
}

export const PumpKpiCard: React.FC<PumpKpiCardProps> = ({
  pumpName,
  solarKw,
  flowRate,
  status,
  onPressAction,
}) => {
  return (
    <View style={styles.card}>
      <View style={styles.header}>
        <Text style={styles.title}>{pumpName}</Text>
        <View style={styles.badge}>
          <Text style={styles.badgeText}>{status}</Text>
        </View>
      </View>

      <View style={styles.metricsRow}>
        <View style={styles.metricBoxSolar}>
          <Text style={styles.metricLabel}>Công Suất Pin</Text>
          <Text style={styles.metricValueSolar}>{solarKw} kW</Text>
        </View>

        <View style={styles.metricBoxWater}>
          <Text style={styles.metricLabel}>Lưu Lượng</Text>
          <Text style={styles.metricValueWater}>{flowRate} m³/h</Text>
        </View>
      </View>

      <TouchableOpacity 
        style={styles.actionButton} 
        onPress={onPressAction}
        accessible={true}
        accessibilityLabel="Khởi động hoặc điều khiển máy bơm"
      >
        <Text style={styles.actionText}>Điều Khiển Trạm Bơm</Text>
      </TouchableOpacity>
    </View>
  );
};

const styles = StyleSheet.create({
  card: {
    backgroundColor: colors.surface.card,
    borderRadius: radius.md,
    padding: spacing[16],
    borderWidth: 1,
    borderColor: colors.border.subtle,
  },
  header: { flexDirection: 'row', justifyContent: 'space-between', alignItems: 'center' },
  title: { ...typography.heading4, color: colors.text.primary },
  badge: { backgroundColor: colors.brand.agri[50], paddingHorizontal: 8, paddingVertical: 4, borderRadius: 999 },
  badgeText: { fontSize: 12, fontWeight: '600', color: colors.brand.agri[500] },
  metricsRow: { flexDirection: 'row', gap: spacing[12], marginTop: spacing[16] },
  metricBoxSolar: { flex: 1, backgroundColor: colors.brand.solar[50], padding: 12, borderRadius: radius.sm },
  metricBoxWater: { flex: 1, backgroundColor: colors.brand.sky[50], padding: 12, borderRadius: radius.sm },
  metricLabel: { ...typography.caption, color: colors.text.secondary },
  metricValueSolar: { ...typography.displayL, color: colors.brand.solar[500], marginTop: 4 },

  actionButton: {
    marginTop: spacing[16],
    backgroundColor: colors.brand.agri[500],
    paddingVertical: 14,
    borderRadius: radius.full,
    alignItems: 'center',
    minHeight: 48,
  },
  actionText: { ...typography.button, color: '#FFFFFF' },
});
import 'package:flutter/material.dart';
import 'tokens.dart';

class PumpKpiCard extends StatelessWidget {
  final String pumpName;
  final double solarKw;
  final double flowRate;
  final bool isPumping;
  final VoidCallback? onControlPressed;

  const PumpKpiCard({
    Key? key,
    required this.pumpName,
    required this.solarKw,
    required this.flowRate,
    this.isPumping = true,
    this.onControlPressed,
  }) : super(key: key);

  @override
  Widget build(BuildContext context) {
    return Container(
      padding: const EdgeInsets.all(SolarPumpSpacing.space16),
      decoration: BoxDecoration(
        color: SolarPumpColors.surfaceCard,
        borderRadius: SolarPumpRadius.md,
        border: Border.all(color: SolarPumpColors.neutral200),
      ),
      child: Column(
        crossAxisAlignment: CrossAxisAlignment.stretch,
        children: [
          Row(
            mainAxisAlignment: MainAxisAlignment.between,
            children: [
              Text(pumpName, style: SolarPumpTypography.heading1.copyWith(fontSize: 18)),
              Chip(
                backgroundColor: SolarPumpColors.agri50,
                label: Text(
                  isPumping ? 'ĐANG CHẠY' : 'DỪNG',
                  style: const TextStyle(color: SolarPumpColors.agri500, fontSize: 11, fontWeight: FontWeight.bold),
                ),
              ),
            ],
          ),
          const SizedBox(height: SolarPumpSpacing.space16),
          Row(
            children: [
              Expanded(
                child: _buildMetricTile(
                  label: 'Công Suất Pin',
                  value: '${solarKw.toStringAsFixed(1)} kW',
                  bgColor: SolarPumpColors.solar50,
                  textColor: SolarPumpColors.solar500,
                ),
              ),
              const SizedBox(width: SolarPumpSpacing.space12),
              Expanded(
                child: _buildMetricTile(
                  label: 'Lưu Lượng',
                  value: '${flowRate.toStringAsFixed(1)} m³/h',
                  bgColor: SolarPumpColors.sky50,
                  textColor: SolarPumpColors.sky500,
                ),
              ),
            ],
          ),
          const SizedBox(height: SolarPumpSpacing.space16),
          ElevatedButton(
            style: ElevatedButton.styleFrom(
              backgroundColor: SolarPumpColors.agri500,
              minimumSize: const Size.fromHeight(48), // 48pt Touch Target
              shape: const StadiumBorder(),
            ),
            onPressed: onControlPressed,
            child: const Text('Điều Khiển Trạm Bơm', style: TextStyle(color: Colors.white, fontWeight: FontWeight.bold)),
          ),
        ],
      ),
    );
  }

  Widget _buildMetricTile({required String label, required String value, required Color bgColor, required Color textColor}) {
    return Container(
      padding: const EdgeInsets.all(SolarPumpSpacing.space12),
      decoration: BoxDecoration(color: bgColor, borderRadius: SolarPumpRadius.sm),
      child: Column(
        crossAxisAlignment: CrossAxisAlignment.start,
        children: [
          Text(label, style: const TextStyle(fontSize: 12, color: SolarPumpColors.neutral600)),
          const SizedBox(height: 4),
          Text(value, style: TextStyle(fontSize: 22, fontWeight: FontWeight.bold, color: textColor)),
        ],
      ),
    );
  }
}
